cmake_minimum_required(VERSION 3.24)
project(FPV_GCS VERSION 0.1.10 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Test Network)
find_package(SDL3 3.2 CONFIG REQUIRED)
set(GSTREAMER_ROOT "C:/gstreamer/1.0/msvc_x86_64" CACHE PATH "GStreamer SDK")
foreach(component gstreamer gstvideo gstbase gobject glib)
  find_library(GST_${component} NAMES ${component}-1.0 ${component}-2.0
    PATHS "${GSTREAMER_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
endforeach()
qt_standard_project_setup()
add_library(fpv_core src/channel.cpp src/controller.cpp src/profile.cpp src/learning.cpp src/rawhid.cpp src/udp.cpp)
target_include_directories(fpv_core PUBLIC src)
target_link_libraries(fpv_core PUBLIC Qt6::Core Qt6::Network SDL3::SDL3 hid user32)
qt_add_executable(FPV_GCS WIN32 src/main.cpp src/window.cpp src/mappingdialog.cpp src/video.cpp src/video.h src/capture.cpp src/capture.h assets/app.rc)
set_source_files_properties(assets/app.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/fatum.ico")
qt_add_resources(FPV_GCS "video_assets" PREFIX "/video" BASE "assets" FILES assets/fatum.jpg)
qt_add_resources(FPV_GCS "app_assets" PREFIX "/app" BASE "assets" FILES assets/fatum.ico)
target_include_directories(FPV_GCS PRIVATE assets)
target_link_libraries(FPV_GCS PRIVATE fpv_core Qt6::Widgets winmm)
target_include_directories(FPV_GCS PRIVATE "${GSTREAMER_ROOT}/include/gstreamer-1.0"
  "${GSTREAMER_ROOT}/include/glib-2.0" "${GSTREAMER_ROOT}/lib/glib-2.0/include")
target_link_libraries(FPV_GCS PRIVATE ${GST_gstreamer} ${GST_gstvideo} ${GST_gstbase} ${GST_gobject} ${GST_glib})
set_target_properties(FPV_GCS PROPERTIES OUTPUT_NAME "FPV GCS")
add_executable(fpv_probe src/probe.cpp)
target_link_libraries(fpv_probe PRIVATE fpv_core)
enable_testing()
add_executable(fpv_tests tests/tests.cpp)
target_link_libraries(fpv_tests PRIVATE fpv_core Qt6::Test)
add_test(NAME channels_profiles_and_sdl COMMAND fpv_tests)
add_executable(fpv_capture_tests tests/capture_tests.cpp src/capture.cpp src/capture.h)
target_include_directories(fpv_capture_tests PRIVATE src)
target_link_libraries(fpv_capture_tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Test)
add_test(NAME capture_format_and_scroll COMMAND fpv_capture_tests)
foreach(target FPV_GCS fpv_probe fpv_tests)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:${target}>)
endforeach()
